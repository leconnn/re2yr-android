cmake_minimum_required(VERSION 3.22)

project(re2yr VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
	set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()

if(CMAKE_SIZEOF_VOID_P EQUAL 4)
	message(FATAL_ERROR "re2yr needs a 64-bit target (32-bit x86 would use x87 maths and break determinism)")
endif()

if(ANDROID)
	set(RA_TESTS_DEFAULT OFF)
else()
	set(RA_TESTS_DEFAULT ON)
endif()
option(RA_BUILD_TESTS "Build the unit tests" ${RA_TESTS_DEFAULT})
option(RA_BUILD_APP "Build the game executable / Android library" ON)
if(ANDROID)
	set(RA_TOOLS_DEFAULT OFF)
else()
	set(RA_TOOLS_DEFAULT ON)
endif()
option(RA_BUILD_TOOLS "Build the desktop developer tools (asset viewer)" ${RA_TOOLS_DEFAULT})

include(cmake/CompilerFlags.cmake)

# Third-party libraries (git submodules in external/)
if(RA_BUILD_APP)
	if(ANDROID)
		set(SDL_SHARED ON CACHE BOOL "" FORCE)
		set(SDL_STATIC OFF CACHE BOOL "" FORCE)
	else()
		set(SDL_SHARED OFF CACHE BOOL "" FORCE)
		set(SDL_STATIC ON CACHE BOOL "" FORCE)
	endif()
	set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
	set(SDL_TESTS OFF CACHE BOOL "" FORCE)
	set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
	set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
	add_subdirectory(external/SDL EXCLUDE_FROM_ALL)
	include(cmake/imgui.cmake)
endif()

add_subdirectory(engine)

if(RA_BUILD_TOOLS AND RA_BUILD_APP)
	add_subdirectory(tools)
endif()

if(RA_BUILD_TESTS)
	enable_testing()
	set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
	set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
	set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
	add_subdirectory(external/googletest EXCLUDE_FROM_ALL)
	add_subdirectory(tests)
endif()
